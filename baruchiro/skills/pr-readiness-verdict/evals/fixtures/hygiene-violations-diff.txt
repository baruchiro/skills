+import { readFileSync, existsSync } from 'fs';
+import path from 'path';
+
+const EXPORT_ROOT = '/data/exports';
+
+export async function exportAttachment(relativePath: string): Promise<Buffer> {
+  const targetPath = path.resolve(EXPORT_ROOT, relativePath);
+
+  if (!existsSync(targetPath)) {
+    throw new Error('Attachment not found');
+  }
+
+  return readFileSync(targetPath);
+}
